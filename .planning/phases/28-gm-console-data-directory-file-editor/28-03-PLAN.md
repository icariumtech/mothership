    Create FileEditorView.tsx:

    Imports: React (useState, useEffect, useCallback, useRef), Editor from '@monaco-editor/react', type { editor } from 'monaco-editor', antd (Button, Tooltip), @ant-design/icons (SaveOutlined, FolderOpenOutlined, FileOutlined, ExclamationCircleOutlined), DataFileTree from '../DataFileTree', gmConsoleApi from '@/services/gmConsoleApi', './FileEditorView.css'.

    State:
      selectedPath: string | null — currently open file path
      content: string — current editor content (may have unsaved changes)
      savedContent: string — the content as last loaded or saved (to compute isDirty)
      isSaving: boolean — true while PUT in flight
      saveFlash: boolean — true for 150ms after successful save
      errorMessage: string | null — validation error from backend (null = hidden)
      errorDismissTimer: useRef<ReturnType<typeof setTimeout> | null>

    isDirty: boolean = content !== savedContent (derived, no separate state)

    editorRef: useRef<editor.IStandaloneCodeEditor | null> — stores editor instance for Ctrl+S registration

    handleSave: useCallback that:
      1. If !selectedPath or !isDirty or isSaving: return early
      2. setIsSaving(true)
      3. try: await gmConsoleApi.writeDataFile(selectedPath, content)
         setErrorMessage(null); setSavedContent(content); setSaveFlash(true); setTimeout(() => setSaveFlash(false), 150)
      4. catch: extract error message from err.response?.data?.detail || err.response?.data?.error || 'Save failed'; setErrorMessage(msg)
         Clear any existing errorDismissTimer; start new 8s timer: setTimeout(() => setErrorMessage(null), 8000)
      5. finally: setIsSaving(false)

    handleSelectFile: useCallback that:
      1. Sets selectedPath(path)
      2. Saves path to localStorage.setItem('janus_file_editor_last_path', path)
      3. Resets content and savedContent to '' and errorMessage to null
      4. Loads file: try { const text = await gmConsoleApi.readDataFile(path); setContent(text); setSavedContent(text) } catch { setErrorMessage('Failed to load file') }

    localStorage restore: useEffect on mount, read localStorage.getItem('janus_file_editor_last_path'); if truthy, call handleSelectFile with that path.

    onEditorMount: (editorInstance, monaco) => {
      editorRef.current = editorInstance;
      editorInstance.addCommand(monaco.KeyMod.CtrlCmd | monaco.KeyCode.KeyS, () => {
        handleSaveRef.current?.()  — use a ref to avoid stale closure
      });
    }
    handleSaveRef: useRef — updated every render: handleSaveRef.current = handleSave

    Language detection: function getLanguage(path: string): string
      const ext = path.split('.').pop()?.toLowerCase() || '';
      if (ext === 'yaml' || ext === 'yml') return 'yaml';
      if (ext === 'md') return 'markdown';
      return 'plaintext';

    isEditable: (path: string) => getLanguage(path) !== 'plaintext'
    isImage: (path: string) => ['.png','.jpg','.jpeg','.svg','.gif','.webp'].some(e => path.toLowerCase().endsWith(e))

    Breadcrumb rendering: split selectedPath by '/' → directories = all but last, filename = last. Render:

    Save button rendering: hidden when no file selected or file is not editable; otherwise:
      <Tooltip title="Save (Ctrl+S)" placement="left">
        <Button
          size="small"
          icon={<SaveOutlined />}
          loading={isSaving}
          disabled={!isDirty && !isSaving}
          onClick={handleSave}
          style={{
            borderColor: saveFlash ? '#2a5a2a' : (isDirty ? '#c9a050' : '#4a8b8b'),
            color: saveFlash ? '#2a5a2a' : (isDirty ? '#c9a050' : '#4a8b8b'),
            background: saveFlash ? 'rgba(42,90,42,0.3)' : 'transparent',
            opacity: isDirty ? 1 : 0.7,

               <div style={{fontSize:11, fontWeight:600, color:'#c0392b', letterSpacing:'1px'}}>VALIDATION ERROR</div>
               <div style={{fontSize:12, color:'#7ab8b8'}}>{errorMessage}</div>
             </div>
           </div>
         </div>
      2. If no file selected: empty state panel (FolderOpenOutlined 48px + SELECT A FILE heading + body text)
      3. If file selected and isImage(selectedPath): <img> tag with src={'/data/' + selectedPath} max-width 100% max-height 80% object-fit contain, plus filename caption
      4. If file selected and not editable and not image: binary panel (FileOutlined 48px + BINARY FILE — CANNOT EDIT + path)
      5. If file selected and editable: <div className="gm-file-editor-view__monaco"> wrapping <Editor theme="vs-dark" language={getLanguage(selectedPath)} value={content} onChange={(val) => setContent(val ?? '')} onMount={onEditorMount} options={{ minimap:{enabled:false}, scrollBeyondLastLine:false, wordWrap: getLanguage(selectedPath) === 'markdown' ? 'on' : 'off', fontFamily:"'Cascadia Code','Courier New',monospace", fontSize:14 }} />
