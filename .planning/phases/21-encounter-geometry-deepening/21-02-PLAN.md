```ts
// Phase 21: canonical Door model (top-level, replaces nested DoorDef)

/** Canonical Door — used by roomGeometry, mapView, renderer. */
export interface Door {
  id: string;
  x: number;          // grid coords (door center)
  y: number;
  angle: number;      // degrees; any value, supports diagonal doors
  width: number;      // grid cells; default 1; fractional allowed
  roomA: string;
  roomB: string | null;   // null = exterior (e.g. airlock to space)
  type: DoorType;
  status: DoorStatus;
}

/** Authored relational door — primary YAML form. */
export interface AuthoredDoorRel {
  id?: string;
  rooms: [string, string] | [string];   // single-element = exterior
  along: number;                          // 0..1 fraction along shared edge
  width?: number;                         // default 1
  type: DoorType;
  status: DoorStatus;
}

/** Authored position-override door — escape hatch for fine control. */
export interface AuthoredDoorPos {
  id?: string;
  rooms: [string, string] | [string];
  position: { x: number; y: number; angle: number };
  width?: number;                         // default 1
  type: DoorType;
  status: DoorStatus;
}

export type AuthoredDoor = AuthoredDoorRel | AuthoredDoorPos;

export class DoorNormalizationError extends Error {
  constructor(message: string, public authored: AuthoredDoor) {
    super(message);
    this.name = 'DoorNormalizationError';
  }
}
```

Do NOT modify or remove any of:

- `GridRoom.doors?: DoorDef[]`
- Any other existing exports

```ts
  return 'position' in d;
```

Create `src/components/domain/encounter/doors/doorNormalizer.ts`. Export:

```ts
export function normalizeDoor(
  authored: AuthoredDoor,
  rooms: GridRoom[],
  index: number,
): Door

export function normalizeDoors(
  authored: AuthoredDoor[],
  rooms: GridRoom[],
): Door[]
```

Implementation requirements:

**Shared-edge resolution.** Compute the shared boundary between `roomA` and `roomB` (or roomA and exterior if `roomB === null`). For rect-only rooms, use the wall-segment algorithm (cell edges that appear on exactly one side of one room AND on the other). For polygon/circle rooms, find shared polygon edges (segment overlap).

If multiple disjoint shared edges exist, pick the longest by default; document this. Future: optional `edge: number` discriminator in authored shape.

For exterior doors (`roomB === null`), the "shared edge" is the room's own exterior boundary; `along` walks that boundary.

**B-rel resolution.** Given resolved shared edge `[(x1,y1), (x2,y2)]`:
- Door center: `((1-along)·x1 + along·x2, (1-along)·y1 + along·y2)`
- Door angle: derived from edge normal — perpendicular to edge direction. For axis-aligned edges, this is 0 (horizontal slot, EW wall axis) or 90 (vertical slot, NS wall axis). For diagonal edges, it's the perpendicular angle in degrees.

**B-pos validation.** Given `{x, y, angle}`:
- Find shared edge of `(roomA, roomB)`
- Verify `(x, y)` lies on that edge within epsilon = 1e-6
- Throw `DoorNormalizationError` if not
- Use authored `angle` directly (no derivation)

**Width clamp / validation.** Default width = 1. Reject if `width ≤ 0` or `width > shared_edge_length`. Reject if `along < width/(2·edge_length)` or `along > 1 − width/(2·edge_length)` (door would run off edge).

**Overlap detection** (in `normalizeDoors`). Group doors by their shared edge (key on sorted room pair). For each group, sort by `along`, check that `[along - w/(2·edge_length), along + w/(2·edge_length)]` intervals are disjoint.

**Stable id derivation.** When `authored.id` is missing:
- `${roomA}__${roomB}__${index}` for two-room doors (use `index` to disambiguate multiple doors between same pair)
- `${roomA}__exterior__${index}` for exterior doors
- `index` is the door's position in the input array, not the position on the edge

Errors thrown:
- `"door references unknown room '${name}'"`
- `"rooms '${a}' and '${b}' share no boundary"`
- `"door width ${w} exceeds shared edge length ${len}"`
- `"door 'along' value places door off the shared edge"`
- `"door position (${x}, ${y}) does not lie on shared edge of '${a}' and '${b}'"`
- `"doors '${id1}' and '${id2}' overlap on shared edge"`

All wrapped in `DoorNormalizationError` with the authored input attached.

- explicit id → preserved
- two doors between same pair → ids differ by index
